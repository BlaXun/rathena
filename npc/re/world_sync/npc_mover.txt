//===== rAthena Script =======================================
//= World sync: NPC mover
//===== Description: =========================================
//= @npcplace lets a GM move any NPC on the map and keeps the new
//= spot across restarts (permanent server variables, applied at
//= start-up). Stand next to the NPC and type @npcplace:
//=   pick the NPC, then "Move here", a one-cell nudge, or "Reset".
//= @npcplace list   shows every saved position, as lines that can
//=                  go into tools/world_sync/npc_positions.txt
//= Only NPCs that stand on a map can be moved (not across maps).
//============================================================

-	script	NpcMover#ws	-1,{
	end;

OnInit:
	bindatcmd "npcplace", strnpcinfo(3) + "::OnCmd", 60, 99;
	for (.@i = 0; .@i < getarraysize($ws_mv$); .@i++) {
		if ($ws_mv$[.@i] == "")
			continue;
		if (getmapxy(.@m$, .@x, .@y, BL_NPC, $ws_mv$[.@i]) != 0)
			continue;                       // the NPC isn't loaded any more
		movenpc $ws_mv$[.@i], $ws_mvx[.@i], $ws_mvy[.@i];
	}
	end;

OnCmd:
	if (.@atcmd_numparameters > 0 && .@atcmd_parameters$[0] == "list") {
		.@n = 0;
		for (.@i = 0; .@i < getarraysize($ws_mv$); .@i++) {
			if ($ws_mv$[.@i] == "")
				continue;
			getmapxy(.@m$, .@x, .@y, BL_NPC, $ws_mv$[.@i]);
			dispbottom $ws_mv$[.@i] + " -> " + .@m$ + " " + $ws_mvx[.@i] + " " + $ws_mvy[.@i];
			.@n++;
		}
		dispbottom .@n + " NPC(s) moved with @npcplace.";
		end;
	}
	getmapxy(.@pm$, .@px, .@py, BL_PC);
	.@count = getareaunits(BL_NPC, .@pm$, .@px - 3, .@py - 3, .@px + 3, .@py + 3, .@gid);
	.@menu$ = "";
	.@k = 0;
	for (.@i = 0; .@i < .@count; .@i++) {
		.@ex$ = rid2name(.@gid[.@i]);
		if (.@ex$ == "" || .@ex$ == strnpcinfo(3))
			continue;
		.@names$[.@k] = .@ex$;
		explode(.@part$, .@ex$, "#");
		.@menu$ += (.@part$[0] == "" ? .@ex$ : .@part$[0]) + ":";
		.@k++;
	}
	if (.@k == 0) {
		dispbottom "No NPC within 3 cells. Stand next to the one you want to move.";
		end;
	}
	mes "[NPC Mover]";
	mes "Which NPC?";
	.@pick = select(.@menu$ + "Cancel") - 1;
	if (.@pick >= .@k)
		close;
	.@ex$ = .@names$[.@pick];
	getmapxy(.@m$, .@x, .@y, BL_NPC, .@ex$);
	next;
	mes "[NPC Mover]";
	mes "^0000FF" + .@ex$ + "^000000 is at " + .@x + "," + .@y + ".";
	switch (select("Move it where I stand:Nudge north:Nudge south:Nudge west:Nudge east:Reset to its first spot:Cancel")) {
	case 1: .@nx = .@px; .@ny = .@py; break;
	case 2: .@nx = .@x; .@ny = .@y + 1; break;
	case 3: .@nx = .@x; .@ny = .@y - 1; break;
	case 4: .@nx = .@x - 1; .@ny = .@y; break;
	case 5: .@nx = .@x + 1; .@ny = .@y; break;
	case 6:
		for (.@i = 0; .@i < getarraysize($ws_mv$); .@i++) {
			if ($ws_mv$[.@i] == .@ex$) {
				movenpc .@ex$, $ws_ox[.@i], $ws_oy[.@i];
				$ws_mv$[.@i] = "";
				mes "Back at " + $ws_ox[.@i] + "," + $ws_oy[.@i] + ".";
				close;
			}
		}
		mes "It hasn't been moved.";
		close;
	default:
		close;
	}
	if (!checkcell(.@m$, .@nx, .@ny, cell_chkpass)) {
		mes "Can't stand there: that cell isn't walkable.";
		close;
	}
	movenpc .@ex$, .@nx, .@ny;
	// remember it (and its first spot, for Reset)
	.@slot = -1;
	for (.@i = 0; .@i < getarraysize($ws_mv$); .@i++) {
		if ($ws_mv$[.@i] == .@ex$) {
			.@slot = .@i;
			break;
		}
		if (.@slot < 0 && $ws_mv$[.@i] == "")
			.@free = .@i + 1;
	}
	if (.@slot < 0) {
		.@slot = .@free ? .@free - 1 : getarraysize($ws_mv$);
		$ws_mv$[.@slot] = .@ex$;
		$ws_ox[.@slot] = .@x;
		$ws_oy[.@slot] = .@y;
	}
	$ws_mvx[.@slot] = .@nx;
	$ws_mvy[.@slot] = .@ny;
	mes "Moved to " + .@nx + "," + .@ny + ". It stays there after restarts.";
	close;
}
