1@4drk,264,43,3	script	Oscar#dk1	4_ED_OSCAR,{
	cutin "OSCAR03.bmp",2;
	switch('dk_step) {
	case 0:
		mes "[Oscar]";
		mes "Welcome to the Dragon's Test.";
		mes "The first trial is the ^0000ffTrial of Life^000000.";
		next;
		mes "[Oscar]";
		mes "Along this cave lie ^0000ff7 Dragon Eggs^000000 whose life has grown cold.";
		mes "Share your own strength with them until each one wakes again.";
		next;
		mes "[Oscar]";
		mes "It will drain your HP and SP, and an egg may need several tries.";
		mes "Rest whenever you must.";
		next;
		switch( select( "I'm ready!", "Give up the test." ) ) {
		case 1:
			'dk_step = 1;
			mes "[Oscar]";
			mes "Then go. The eggs are waiting.";
			close3;
		case 2:
			callsub S_GiveUp;
		}
		end;
	case 1:
		mes "[Oscar]";
		mes "Eggs revived: ^0000ff" + 'dk_eggs + " / 7^000000.";
		mes "Keep going, the rest are further along the cave.";
		next;
		if (select( "Continue.", "Give up the test." ) == 2)
			callsub S_GiveUp;
		close3;
	default:
		mes "[Oscar]";
		mes "The eggs are warm again. You did well.";
		close3;
	}
	end;

S_GiveUp:
	mes "[Oscar]";
	mes "Giving up ends this test. You'll need a rest before you can try again.";
	mes "Are you sure?";
	next;
	if (select( "Stay.", "Leave the test." ) == 1)
		close3;
	.@id = instance_id();
	setquest 12509;
	cutin "",255;
	warp "gef_fild08",54,98;
	instance_destroy .@id;
	end;

OnInstanceInit:
	'dk_map$ = instance_mapname("1@4drk");
	'dk_step = 0;
	'dk_eggs = 0;
	'dk_fails = 0;
	// Fountains of Life (trial 2), in order.
	setarray 'dk_fx[1], 19, 25, 43, 65, 65, 68, 78,110,133,136;
	setarray 'dk_fy[1],100,126,141,152,181,208,231,232,242,271;
	// Spawn points of Leticia's soldiers (trial 3).
	setarray 'dk_sx[0], 47, 59, 70, 92,113,122,128,145,160,167;
	setarray 'dk_sy[0], 44, 44, 44, 46, 49, 52, 57, 64, 73, 78;
	setmapflag 'dk_map$, MF_NOBASEEXP;
	setmapflag 'dk_map$, MF_NOJOBEXP;
	setmapflag 'dk_map$, MF_NOMOBLOOT;
	setmapflag 'dk_map$, MF_NOMVPLOOT;
	end;
}

1@4drk,264,50,0	script	Dragon Egg#dk1	4_DRAGON_EGG,{
	if ('dk_step != 1) {
		mes "- A cold Dragon Egg. -";
		mes "- Oscar may know what to do with it. -";
		close;
	}
	mes "- The egg is cold, but something inside still stirs faintly. -";
	mes "- You could share your life force with it. -";
	next;
	while (true) {
		if (select( "Share your life force.", "Step back." ) == 2)
			close;
		if (Hp * 100 / MaxHp < 20 || Sp * 100 / MaxSp < 20) {
			mes "- You are too exhausted to give any more. -";
			mes "- Rest or recover first. -";
			close;
		}
		.@hp = rand(5,15);
		.@sp = rand(5,15);
		percentheal -.@hp, -.@sp;
		specialeffect EF_HEAL2;
		if (rand(100) < 30)
			break;
		mes "- The egg flickers for a moment, then goes still again. -";
		next;
	}
	'dk_eggs++;
	specialeffect EF_RESURRECTION;
	mes "- The shell warms under your hands. The egg is alive again! -";
	mes "- Eggs revived: ^0000ff" + 'dk_eggs + " / 7^000000 -";
	if ('dk_eggs < 7) {
		close2;
		disablenpc();
		end;
	}
	next;
	'dk_step = 2;
	mes "- You hear Oscar's voice from somewhere far away. -";
	mes "- \"Well done. Come, the next trial waits for you.\" -";
	close2;
	disablenpc();
	warp 'dk_map$,19,72;
	end;
}

1@4drk,1,1,0	script	#dk_escort	-1,{
	end;

OnStart:
	stopnpctimer;
	.@npc$ = instance_npcname("#dk_escort");
	killmonster 'dk_map$, .@npc$ + "::OnBoneDead";
	killmonster 'dk_map$, .@npc$ + "::OnGoldDead";
	for (.@i = 1; .@i <= 10; .@i++)
		disablenpc instance_npcname("Fountain of Life#dk" + .@i);
	'dk_step = 21;
	'dk_fountain = 1;
	monster 'dk_map$,19,78,"Golden Acidus","4JOB_ACIDUS",1, .@npc$ + "::OnGoldDead";
	'dk_gold = $@mobid[0];
	// The dragon arrives dying: a tenth of its HP, healed at each fountain.
	getunitdata 'dk_gold, .@d;
	setunitdata 'dk_gold, UMOB_HP, .@d[UMOB_MAXHP] / 10;
	callsub S_Fountain;
	initnpctimer;
	end;

// Shows the current fountain and its guard.
// After 3 failures the guards are no longer spawned.
S_Fountain:
	enablenpc instance_npcname("Fountain of Life#dk" + 'dk_fountain);
	if ('dk_fails < 3) {
		monster 'dk_map$,'dk_fx['dk_fountain],'dk_fy['dk_fountain],"Bone Acidus","BONE_ACIDUS",1, instance_npcname("#dk_escort") + "::OnBoneDead";
		setunitdata $@mobid[0], UMOB_MAXHP, 100000;
		setunitdata $@mobid[0], UMOB_HP, 100000;
	}
	mapannounce 'dk_map$, "Oscar : Fountain of Life " + 'dk_fountain + " is glowing. Lead the Golden Acidus there!", bc_map, 0xFFFF00;
	return;

OnTimer1000:
	if ('dk_step != 21)
		end;
	if (unitexists('dk_gold) == false) {
		initnpctimer;
		end;
	}
	if (mobcount('dk_map$, instance_npcname("#dk_escort") + "::OnBoneDead") > 0) {
		initnpctimer;
		end;
	}
	getunitdata 'dk_gold, .@d;
	.@f = 'dk_fountain;
	if (distance(.@d[UMOB_X], .@d[UMOB_Y], 'dk_fx[.@f], 'dk_fy[.@f]) > 3) {
		initnpctimer;
		end;
	}
	specialeffect EF_HEAL2, AREA, instance_npcname("Fountain of Life#dk" + .@f);
	// Not min(): it would read .@d[UMOB_MAXHP] as the rest of the array.
	.@hp = .@d[UMOB_HP] + .@d[UMOB_MAXHP] / 10;
	if (.@f == 10 || .@hp > .@d[UMOB_MAXHP])
		.@hp = .@d[UMOB_MAXHP];
	setunitdata 'dk_gold, UMOB_HP, .@hp;
	disablenpc instance_npcname("Fountain of Life#dk" + .@f);
	mapannounce 'dk_map$, "The Golden Acidus drinks from the Fountain of Life. (" + .@f + " / 10)", bc_map, 0x00FF99;
	if (.@f < 10) {
		'dk_fountain++;
		callsub S_Fountain;
		initnpctimer;
		end;
	}
	'dk_step = 30;
	mapannounce 'dk_map$, "Oscar : Its strength is back. The Trial of the Protector is passed!", bc_map, 0xFFFF00;
	sleep 3000;
	killmonster 'dk_map$, instance_npcname("#dk_escort") + "::OnGoldDead";
	sleep 2000;
	mapannounce 'dk_map$, "??? : So the little one trusts you. Then let me see what else you have.", bc_map, 0xFF99CC;
	sleep 3000;
	enablenpc instance_npcname("Leticia#dk3");
	instance_warpall 'dk_map$,92,46, instance_id();
	end;

OnGoldDead:
	if ('dk_step != 21)
		end;
	stopnpctimer;
	'dk_step = 20;
	'dk_fails++;
	killmonster 'dk_map$, instance_npcname("#dk_escort") + "::OnBoneDead";
	for (.@i = 1; .@i <= 10; .@i++)
		disablenpc instance_npcname("Fountain of Life#dk" + .@i);
	mapannounce 'dk_map$, "Oscar : The Golden Acidus has fallen! Come back to me and start over.", bc_map, 0xFFFF00;
	end;

OnBoneDead:
	end;
}

1@4drk,1,2,0	script	#dk_soldiers	-1,{
	end;

OnStart:
	'dk_step = 31;
	'dk_killed = 0;
	for (.@i = 0; .@i < 10; .@i++)
		callsub S_Spawn;
	mapannounce 'dk_map$, "Leticia : Defeat 50 of my soldiers within 5 minutes!", bc_map, 0xFF99CC;
	initnpctimer;
	end;

S_Spawn:
	.@r = rand(getarraysize('dk_sx));
	monster 'dk_map$,'dk_sx[.@r],'dk_sy[.@r],"Leticia's Soldier","BONE_FERUS",1, instance_npcname("#dk_soldiers") + "::OnSoldierDead";
	setunitdata $@mobid[0], UMOB_MAXHP, 30000;
	setunitdata $@mobid[0], UMOB_HP, 30000;
	return;

OnSoldierDead:
	if ('dk_step != 31)
		end;
	'dk_killed++;
	if ('dk_killed >= 50) {
		stopnpctimer;
		'dk_step = 40;
		killmonster 'dk_map$, instance_npcname("#dk_soldiers") + "::OnSoldierDead";
		mapannounce 'dk_map$, "Leticia : 50 already? Then there is one trial left. Come to me!", bc_map, 0xFF99CC;
		sleep 3000;
		disablenpc instance_npcname("Leticia#dk3");
		enablenpc instance_npcname("Leticia#dk4");
		instance_warpall 'dk_map$,211,165, instance_id();
		end;
	}
	if ('dk_killed % 10 == 0)
		mapannounce 'dk_map$, "Leticia's soldiers defeated: " + 'dk_killed + " / 50", bc_map, 0xFF99CC;
	// 10 at the start and one for each of the first 40 kills: 50 in all.
	if ('dk_killed <= 40)
		callsub S_Spawn;
	end;

OnTimer240000:
	if ('dk_step == 31)
		mapannounce 'dk_map$, "Leticia : One minute left!", bc_map, 0xFF99CC;
	end;

OnTimer300000:
	stopnpctimer;
	if ('dk_step != 31)
		end;
	'dk_step = 32;
	killmonster 'dk_map$, instance_npcname("#dk_soldiers") + "::OnSoldierDead";
	mapannounce 'dk_map$, "Leticia : Time's up! Talk to me if you want another try.", bc_map, 0xFF99CC;
	end;
}

1@4drk,1,3,0	script	#dk_dragon	-1,{
	end;

OnStart:
	if ('dk_step != 40)
		end;
	'dk_step = 41;
	'dk_tick = 0;
	disablenpc instance_npcname("Leticia#dk4");
	monster 'dk_map$,219,211,"Dragon Queen Leticia","4JOB_LETICIA",1, instance_npcname("#dk_dragon") + "::OnDragonDead";
	'dk_dragon = $@mobid[0];
	initnpctimer;
	end;

// Every 30 seconds Leticia is covered by a green aura for 10 seconds
// and takes only 10% of the damage.
OnTimer1000:
	if ('dk_step != 41)
		end;
	if (unitexists('dk_dragon) == false) {
		initnpctimer;
		end;
	}
	getunitdata 'dk_dragon, .@d;
	if (.@d[UMOB_HP] <= .@d[UMOB_MAXHP] / 100 * 91) {
		callsub S_Passed;
		end;
	}
	'dk_tick++;
	if ('dk_tick % 30 == 15) {
		setunitdata 'dk_dragon, UMOB_DAMAGETAKEN, 10;
		unittalk 'dk_dragon, "Leticia : Let's see you get through this!";
		mapannounce 'dk_map$, "A green aura covers Leticia. Your attacks barely reach her!", bc_map, 0x00FF00;
	} else if ('dk_tick % 30 == 25) {
		setunitdata 'dk_dragon, UMOB_DAMAGETAKEN, 100;
		mapannounce 'dk_map$, "The green aura around Leticia fades.", bc_map, 0x00FF00;
	}
	initnpctimer;
	end;

OnDragonDead:
	if ('dk_step == 41)
		callsub S_Passed;
	end;

S_Passed:
	stopnpctimer;
	'dk_step = 50;
	killmonster 'dk_map$, instance_npcname("#dk_dragon") + "::OnDragonDead";
	mapannounce 'dk_map$, "Leticia : Enough, enough! Come and talk to me.", bc_map, 0xFF99CC;
	sleep 2000;
	enablenpc instance_npcname("Leticia#dk4");
	instance_warpall 'dk_map$,211,165, instance_id();
	end;
}
